/* ========================================================================== JP M3U GENERATOR V2 - IndexedDB Engine (Zero LocalStorage Overflow) ========================================================================== */ const DB_NAME = 'JP_M3U_DB'; const DB_VERSION = 1; let db = null; // Persistent Storage Active (যাতে ব্রাউজার নিজে থেকে ডেটা ডিলিট না করে) if (navigator.storage && navigator.storage.persist) { navigator.storage.persist().then(granted => { console.log("Persistent Storage Status:", granted ? "Permanent" : "Temporary"); }); } // 1. IndexedDB ইনিশিয়ালাইজেশন function initDB() { return new Promise((resolve, reject) => { const request = indexedDB.open(DB_NAME, DB_VERSION); request.onupgradeneeded = (e) => { const database = e.target.result; if (!database.objectStoreNames.contains('playlists')) { database.createObjectStore('playlists', { keyPath: 'id' }); } if (!database.objectStoreNames.contains('logos')) { database.createObjectStore('logos', { keyPath: 'id' }); } }; request.onsuccess = (e) => { db = e.target.result; resolve(db); }; request.onerror = (e) => { console.error("IndexedDB Open Error:", e.target.error); reject(e.target.error); }; }); } // 2. সেফ ডেটা সেভ (IndexedDB Priority - নো পপআপ) async function saveDataToIDB(storeName, data) { if (!db) await initDB(); return new Promise((resolve, reject) => { try { const tx = db.transaction(storeName, 'readwrite'); const store = tx.objectStore(storeName); const request = store.put(data); request.onsuccess = () => resolve(true); request.onerror = (e) => { console.error(`IDB Save Error [${storeName}]:`, e.target.error); resolve(false); }; } catch (err) { console.error("IDB Transaction Fail:", err); resolve(false); } }); } // 3. ডেটা ফেচিং ফাংশন async function getDataFromIDB(storeName, key) { if (!db) await initDB(); return new Promise((resolve) => { try { const tx = db.transaction(storeName, 'readonly'); const store = tx.objectStore(storeName); const request = store.get(key); request.onsuccess = () => resolve(request.result ? request.result.data : null); request.onerror = () => resolve(null); } catch (e) { resolve(null); } }); } // 4. অ্যাপ স্টেট ও গ্লোবাল ভ্যারিয়েবল let playlists = []; let currentPlaylistId = 'default'; let currentPage = 1; let pageSize = 50; let selectedChannels = new Set(); // 5. পেজ লোড ও ইউআই লজিক document.addEventListener("DOMContentLoaded", async () => { await initDB(); // ট্যাবিং লজিক const navButtons = document.querySelectorAll('.nav button'); const tabs = document.querySelectorAll('.tab'); navButtons.forEach(btn => { btn.addEventListener('click', () => { const targetTab = btn.getAttribute('data-tab'); navButtons.forEach(b => b.classList.remove('active')); tabs.forEach(t => t.classList.remove('active')); btn.classList.add('active'); const activeTab = document.getElementById(targetTab); if(activeTab) activeTab.classList.add('active'); // টাইটেল আপডেট const titleMap = { 'editor': 'Playlist Editor', 'create': 'Create Channel', 'logo': 'Logo Manager', 'builder': 'Playlist Builder', 'tools': 'M3U Tools', 'export': 'Export Options' }; document.getElementById('title').innerText = titleMap[targetTab] || 'Playlist Editor'; }); }); // প্লেলিস্ট লোড await loadInitialData(); }); // ৬. ব্যাকগ্রাউন্ড সেভ ও প্লেলিস্ট সিঙ্ক async function loadInitialData() { const savedData = await getDataFromIDB('playlists', currentPlaylistId); if (savedData) { playlists = savedData; } else { playlists = [{ id: 1, name: "Sample Channel", group: "General", url: "", logo: "" }]; await saveDataToIDB('playlists', { id: currentPlaylistId, data: playlists }); } renderPlaylist(); } // ৭. নিরাপদে চ্যানেল / লোগো যুক্ত বা সেভ করা (এখানেই আগে alert ছিল যা সরানো হয়েছে) async function saveChannelOrLogo(itemData) { // LocalStorage সম্পূর্ণ বাদ দিয়ে ব্যাকগ্রাউন্ডে IndexedDB-তে সেভ করা const status = await saveDataToIDB('logos', { id: Date.now().toString(), data: itemData }); if(status) { console.log("Logo/File saved successfully in IndexedDB."); } } // ৮. চ্যানেল রেন্ডারিং ও UI কন্ট্রোল function renderPlaylist() { const listContainer = document.getElementById('list'); if (!listContainer) return; if (playlists.length === 0) { listContainer.innerHTML = `
কোনো চ্যানেল পাওয়া যায়নি
`; updateStats(); return; } let html = ''; const start = (currentPage - 1) * pageSize; const end = start + pageSize; const pagedItems = playlists.slice(start, end); pagedItems.forEach((ch, index) => { const realIndex = start + index; html += `
⋮⋮
${ch.name || 'Untitled Channel'}
${ch.group || 'Unassigned'}
`; }); listContainer.innerHTML = html; updateStats(); updatePagination(); } function updateStats() { document.getElementById('sChannels').innerText = playlists.length; const groups = new Set(playlists.map(p => p.group).filter(Boolean)); document.getElementById('sGroups').innerText = groups.size; const logos = playlists.filter(p => p.logo && p.logo.length > 0); document.getElementById('sLogos').innerText = logos.length; } function updatePagination() { const totalPages = Math.ceil(playlists.length / pageSize) || 1; document.getElementById('pageInfo').innerText = `Page ${currentPage}/${totalPages}`; } function gotoPage(page) { const totalPages = Math.ceil(playlists.length / pageSize) || 1; if (page < 1 || page > totalPages) return; currentPage = page; renderPlaylist(); } function toggleSelAll(checked) { if (checked) { playlists.forEach((_, idx) => selectedChannels.add(idx)); } else { selectedChannels.clear(); } document.getElementById('selCount').innerText = `${selectedChannels.size} সিলেক্ট করা হয়েছে`; renderPlaylist(); } function toggleSelect(idx) { if (selectedChannels.has(idx)) { selectedChannels.delete(idx); } else { selectedChannels.add(idx); } document.getElementById('selCount').innerText = `${selectedChannels.size} সিলেক্ট করা হয়েছে`; } // ৯. ফাইল সেভ/আপডেট ওভাররাইড (কোনো ওয়ার্নিং ছাড়াই সেভ হবে) async function syncChanges() { await saveDataToIDB('playlists', { id: currentPlaylistId, data: playlists }); } function deleteChannel(index) { playlists.splice(index, 1); syncChanges(); renderPlaylist(); } function openEditor() { const name = prompt("চ্যানেলের নাম লিখুন:"); const url = prompt("স্ট্রিম URL লিখুন:"); if (name && url) { playlists.push({ id: Date.now(), name, url, group: "General", logo: "" }); syncChanges(); renderPlaylist(); } }
